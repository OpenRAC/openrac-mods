openrac_add_mod(modern_controls MANIFEST openrac-mod.json SOURCES src/modern_controls.cpp)
target_include_directories(modern_controls PRIVATE src ../keyboard-mouse/src)
if(NOT MSVC)
  target_compile_options(modern_controls PRIVATE -Wall -Wextra)
endif()
if(BUILD_TESTING)
  add_executable(modern_controls_test tests/controls_test.cpp)
  target_include_directories(modern_controls_test PRIVATE src ../keyboard-mouse/src)
  target_link_libraries(modern_controls_test PRIVATE OpenRAC::mod)
  set_target_properties(modern_controls_test PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
  add_test(NAME modern_controls COMMAND modern_controls_test)
endif()

if(BUILD_TESTING AND UNIX)
  add_executable(modern_controls_load_test tests/load_test.cpp)
  target_link_libraries(modern_controls_load_test PRIVATE OpenRAC::mod ${CMAKE_DL_LIBS})
  set_target_properties(modern_controls_load_test PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
  add_dependencies(modern_controls_load_test modern_controls)
  add_test(NAME modern_controls_load COMMAND modern_controls_load_test $<TARGET_FILE:modern_controls>)
endif()

get_target_property(controls_package modern_controls OPENRAC_MOD_PACKAGE_DIR)
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/../LICENSE" "${controls_package}/LICENSE" COPYONLY)
configure_file(README.md "${controls_package}/README.md" COPYONLY)
