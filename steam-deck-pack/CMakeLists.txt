openrac_add_mod(steam_deck_pack MANIFEST openrac-mod.json SOURCES src/steam_deck.cpp)
target_include_directories(steam_deck_pack PRIVATE src)
target_link_libraries(steam_deck_pack PRIVATE controller_icons)
if(MSVC)
  target_compile_options(steam_deck_pack PRIVATE /utf-8)
else()
  target_compile_options(steam_deck_pack PRIVATE -Wall -Wextra)
endif()
if(BUILD_TESTING)
  add_executable(steam_deck_logic_test tests/logic_test.cpp)
  target_include_directories(steam_deck_logic_test PRIVATE src)
  set_target_properties(steam_deck_logic_test PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
  add_test(NAME steam_deck_logic COMMAND steam_deck_logic_test)
  add_executable(steam_deck_load_test tests/load_test.cpp)
  target_link_libraries(steam_deck_load_test PRIVATE OpenRAC::mod controller_icons ${CMAKE_DL_LIBS})
  set_target_properties(steam_deck_load_test PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
  add_dependencies(steam_deck_load_test steam_deck_pack)
  add_test(NAME steam_deck_load COMMAND steam_deck_load_test $<TARGET_FILE:steam_deck_pack>)
  if(UNIX)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME steam_deck_installer COMMAND "${Python3_EXECUTABLE}" -m unittest discover
             -s "${CMAKE_CURRENT_SOURCE_DIR}/tests" -p "test_*.py")
  endif()
endif()
# Include the offline setup utility in every native release archive.
get_target_property(deck_package steam_deck_pack OPENRAC_MOD_PACKAGE_DIR)
file(COPY extras DESTINATION "${deck_package}" PATTERN "__pycache__" EXCLUDE)
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/../LICENSE" "${deck_package}/LICENSE" COPYONLY)
